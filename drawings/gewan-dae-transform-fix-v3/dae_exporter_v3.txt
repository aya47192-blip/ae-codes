
function exportDae(g,family){
 function trs(m){let sx=Math.hypot(m[0],m[1],m[2]),sy=Math.hypot(m[4],m[5],m[6]),sz=Math.hypot(m[8],m[9],m[10]),rz=Math.atan2(m[1]/sx,m[0]/sx),rx=Math.atan2(m[6]/sy,m[10]/sz);let cz=Math.cos(rz),ss=Math.sin(rz),cx=Math.cos(rx),sn=Math.sin(rx),r=[cz*sx,ss*sx,0,0,-ss*cx*sy,cz*cx*sy,sn*sy,0,ss*sn*sz,-cz*sn*sz,cx*sz,0,m[12],m[13],m[14],1];if(r.some((v,i)=>Math.abs(v-m[i])>1e-8))throw Error('TRS decomposition residual');return '<translate>'+[m[12],m[13],m[14]].map(num).join(' ')+'</translate><rotate>0 0 1 '+num(rz*180/Math.PI)+'</rotate><rotate>1 0 0 '+num(rx*180/Math.PI)+'</rotate><scale>'+[sx,sy,sz].map(num).join(' ')+'</scale>';}
 const map=new Map(g.nodes.map(n=>[n.id,n])),used=new Set();function go(id){if(used.has(id))return;used.add(id);for(let ch of map.get(id).ch)go(ch.n);}g.roots[family].forEach(go);
 const names=[...used],nodes=names.map(id=>map.get(id)),materials=[...new Set(nodes.flatMap(n=>n.t.map(t=>t[0])))],esc=s=>s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/"/g,'&quot;'),num=v=>Number(v.toFixed(6)).toString(),out=[];
 out.push('<?xml version="1.0" encoding="utf-8"?><COLLADA xmlns="http://www.collada.org/2005/11/COLLADASchema" version="1.4.1"><asset><contributor><author>Gewan Design Study</author></contributor><created>2026-11-22T12:00:00Z</created><modified>2026-11-22T12:00:00Z</modified><unit name="meter" meter="1"/><up_axis>Z_UP</up_axis></asset>');
 const images=[...new Set(materials.map(m=>g.material_info[m]?.texture).filter(Boolean))];out.push('<library_images>');for(let[i,p]of images.entries())out.push('<image id="image_'+i+'"><init_from>'+p+'</init_from></image>');out.push('</library_images><library_effects>');
 for(let m of materials){let c=g.mats[m],info=g.material_info[m]||{},tex=info.texture,img=images.indexOf(tex);out.push('<effect id="'+m+'_fx"><profile_COMMON>');if(tex)out.push('<newparam sid="'+m+'_surface"><surface type="2D"><init_from>image_'+img+'</init_from></surface></newparam><newparam sid="'+m+'_sampler"><sampler2D><source>'+m+'_surface</source></sampler2D></newparam>');out.push('<technique sid="common"><phong><emission><color>'+((info.emission)?c.slice(0,3).join(' ')+' 1':'0 0 0 1')+'</color></emission><ambient><color>0.2 0.2 0.2 1</color></ambient><diffuse>'+(tex?'<texture texture="'+m+'_sampler" texcoord="UVSET0"/>':'<color>'+[...c.slice(0,3),1].join(' ')+'</color>')+'</diffuse><specular><color>0.15 0.15 0.15 1</color></specular><shininess><float>24</float></shininess>');if(c.length>3)out.push('<transparent opaque="A_ONE"><color>1 1 1 '+c[3]+'</color></transparent><transparency><float>1</float></transparency>');out.push('</phong></technique></profile_COMMON></effect>');}
 out.push('</library_effects><library_materials>');for(let m of materials)out.push('<material id="'+m+'_material" name="'+m+'"><instance_effect url="#'+m+'_fx"/></material>');out.push('</library_materials><library_geometries>');
 for(let n of nodes){if(!n.t.length)continue;let normals=n.v.map(()=>[0,0,0]),uv=n.v.map(()=>[0,0]),acc=new Map(),keys=[];
 for(let t of n.t){let[a,b,c]=t.slice(1).map(i=>n.v[i]),u=b.map((v,i)=>v-a[i]),v=c.map((q,i)=>q-a[i]),norm=[u[1]*v[2]-u[2]*v[1],u[2]*v[0]-u[0]*v[2],u[0]*v[1]-u[1]*v[0]],major=norm.map(Math.abs).indexOf(Math.max(...norm.map(Math.abs))),axes=[0,1,2].filter(i=>i!==major),tile=g.material_info[t[0]]?.tile_m||1;for(let ix of t.slice(1)){let key=t[0]+'_'+n.v[ix].map(q=>q.toFixed(5)).join(',');keys[ix]=key;let q=acc.get(key)||[0,0,0];acc.set(key,q.map((v,i)=>v+norm[i]));uv[ix]=axes.map(i=>n.v[ix][i]/tile);}}
 for(let i=0;i<n.v.length;i++){let v=acc.get(keys[i])||[0,0,1],l=Math.hypot(...v)||1;normals[i]=v.map(q=>q/l);}
 out.push('<geometry id="'+n.id+'_geo" name="'+esc(n.name)+'"><mesh>');
 for(let[suffix,data,stride,params]of[['pos',n.v,3,['X','Y','Z']],['nor',normals,3,['X','Y','Z']],['uv',uv,2,['S','T']]])out.push('<source id="'+n.id+'_'+suffix+'"><float_array id="'+n.id+'_'+suffix+'_arr" count="'+data.length*stride+'">'+data.map(p=>p.map(num).join(' ')).join(' ')+'</float_array><technique_common><accessor source="#'+n.id+'_'+suffix+'_arr" count="'+data.length+'" stride="'+stride+'">'+params.map(p=>'<param name="'+p+'" type="float"/>').join('')+'</accessor></technique_common></source>');
 out.push('<vertices id="'+n.id+'_vertex"><input semantic="POSITION" source="#'+n.id+'_pos"/></vertices>');
 for(let m of new Set(n.t.map(t=>t[0]))){let ts=n.t.filter(t=>t[0]===m);out.push('<triangles count="'+ts.length+'" material="'+m+'"><input semantic="VERTEX" source="#'+n.id+'_vertex" offset="0"/><input semantic="NORMAL" source="#'+n.id+'_nor" offset="1"/><input semantic="TEXCOORD" source="#'+n.id+'_uv" offset="2" set="0"/><p>'+ts.map(t=>t.slice(1).map(i=>i+' '+i+' '+i).join(' ')).join(' ')+'</p></triangles>');}
 out.push('</mesh></geometry>');}
 out.push('</library_geometries><library_nodes>');let instanceCounter=0;
 for(let n of nodes){out.push('<node id="'+n.id+'" name="'+esc(n.name)+'" type="NODE">');if(n.t.length){out.push('<instance_geometry url="#'+n.id+'_geo"><bind_material><technique_common>');for(let m of new Set(n.t.map(t=>t[0])))out.push('<instance_material symbol="'+m+'" target="#'+m+'_material"><bind_vertex_input semantic="UVSET0" input_semantic="TEXCOORD" input_set="0"/></instance_material>');out.push('</technique_common></bind_material></instance_geometry>');}for(let ch of n.ch)out.push('<node id="child_'+instanceCounter+++'" name="'+esc(map.get(ch.n).name)+'">'+trs(ch.matrix)+'<instance_node url="#'+ch.n+'"/></node>');out.push('</node>');}
 out.push('</library_nodes><library_visual_scenes><visual_scene id="Scene"><node id="Collection" name="GEWAN_'+family+'_Independent_Units">');
 for(let id of g.roots[family]){let record=g.records.find(r=>r.id===id),p=record?.at||[0,0,0];out.push('<node id="position_'+instanceCounter+++'" name="'+esc(map.get(id).name)+'"><translate>'+p.map(num).join(' ')+'</translate><instance_node url="#'+id+'"/></node>');}
 out.push('</node></visual_scene></library_visual_scenes><scene><instance_visual_scene url="#Scene"/></scene></COLLADA>');
 const xml=out.join('');let ids=new Set(),refs=[];for(let match of xml.matchAll(/\bid="([^"]+)"/g)){if(ids.has(match[1]))throw Error('Duplicate COLLADA ID '+match[1]);ids.add(match[1]);}for(let match of xml.matchAll(/(?:url|source|target)="#([^"]+)"/g))refs.push(match[1]);for(let ref of refs)if(!ids.has(ref))throw Error('Dangling XML reference '+ref);
 return{xml,report:{family,root_units:g.roots[family].length,definitions:nodes.length,mesh_definitions:nodes.filter(n=>n.t.length).length,unique_triangles:nodes.reduce((s,n)=>s+n.t.length,0),all_id_references_resolved:true,hierarchy:'library_nodes + instance_node; explicit translate/rotate/scale; no matrix serialization',materials:materials.length,texture_images:images.length,xml_characters:xml.length,desktop_import_tested:false}};
}
return exportDae;
